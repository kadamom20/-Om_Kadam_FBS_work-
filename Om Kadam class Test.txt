Q1)   void main()
   {
    int start,end,i,j,sum=0,count;

    printf("Enter start:");
    scanf("%d",&start);

    printf("Enter end:");
    scanf("%d",&end);

    for(i=start;i<=end;i++)
    {
        count=0;

        for(j=1;j<=i;j++)
        {
            if(i%j==0)
            {
                count++;
            }
        }

        if(count==2)
        {
            printf("%d ",i);
            sum=sum+i;
        }
    }

    printf("\nSum=%d",sum);
}


Q2)    void main()
   {
    char str[50];
    char ch;
    int i;

    printf("Enter a string:");
    scanf("%s",str);

    printf("Enter character to replace:");
    scanf(" %c",&ch);

    for(i=0;str[i]!='\0';i++)
    {
        if(str[i]==ch)
        {
            str[i]='*';
        }
    }

    printf("Updated string=%s",str);
}
